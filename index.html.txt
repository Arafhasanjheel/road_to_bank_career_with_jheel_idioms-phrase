        function renderCards(data) {
            const grid = document.getElementById('cardsGrid');
            const counter = document.getElementById('cardCounter');
            grid.innerHTML = '';
            
            counter.innerText = `Showing ${data.length} Cards`;

            if (data.length === 0) {
                grid.innerHTML = '<div class="no-results">No idioms found matching your search.</div>';
                return;
            }

            data.forEach(item => {
                const card = document.createElement('div');
                card.className = 'card-container';
                card.onclick = (e) => {
                    if (!e.target.classList.contains('audio-btn') && !e.target.parentElement.classList.contains('audio-btn')) {
                        card.classList.toggle('flipped');
                    }
                };

                card.innerHTML = `
                    <div class="card-inner">
                        <div class="card-front">
                            <div class="card-number">#${item.id}</div>
                            <div>
                                <div class="idiom-title">
                                    ${item.phrase}
                                    <button class="audio-btn" onclick="speakText('${item.phrase.replace(/'/g, "\\'")}')" title="Listen">
                                        <i class="fa-solid fa-volume-high"></i>
                                    </button>
                                </div>
                            </div>
                            <div class="flip-hint">
                                <i class="fa-solid fa-rotate"></i> Tap card to see meaning
                            </div>
                        </div>
                        <div class="card-back">
                            <div>
                                <div class="card-number" style="background: rgba(16, 185, 129, 0.1); color: var(--success);">Answer #${item.id}</div>
                                <div class="meaning" style="margin-top: 15px;">
                                    <strong>Meaning:</strong> ${item.meaning}
                                </div>
                                <div class="example">
                                    <strong>Example:</strong> ${item.example}
                                    <button class="audio-btn" onclick="speakText('${item.example.replace(/'/g, "\\'")}')" title="Listen">
                                        <i class="fa-solid fa-volume-high"></i>
                                    </button>
                                </div>
                            </div>
                            <div class="flip-hint" style="color: var(--success);">
                                <i class="fa-solid fa-rotate"></i> Tap card to flip back
                            </div>
                        </div>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        function filterCards() {
            const query = document.getElementById('searchInput').value.toLowerCase();
            const filtered = idiomsData.filter(item => 
                item.phrase.toLowerCase().includes(query) ||
                item.meaning.toLowerCase().includes(query) ||
                item.example.toLowerCase().includes(query)
            );
            renderCards(filtered);
        }

        function speakText(text) {
            if ('speechSynthesis' in window) {
                window.speechSynthesis.cancel();
                const utterance = new SpeechSynthesisUtterance(text);
                utterance.lang = 'en-US';
                utterance.rate = 0.9;
                window.speechSynthesis.speak(utterance);
            }
        }

        function toggleTheme() {
            document.body.classList.toggle('dark-mode');
            const themeText = document.getElementById('theme-text');
            const icon = document.querySelector('.theme-toggle i');
            
            if (document.body.classList.contains('dark-mode')) {
                themeText.innerText = 'Light Mode';
                icon.className = 'fa-solid fa-sun';
            } else {
                themeText.innerText = 'Dark Mode';
                icon.className = 'fa-solid fa-moon';
            }
        }

        // Initial rendering
        renderCards(idiomsData);
    </script>
